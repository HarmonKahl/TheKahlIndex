TEAM COLOR SCRAPER (teamcolorcodes.com/ncaa-color-codes/) # --------------------------------------------------------------------------- def _fetch_html(url: str, timeout: int = 12) -> str: req = urllib.request.Request(url, headers=HEADERS) with urllib.request.urlopen(req, timeout=timeout) as resp: return resp.read().decode("utf-8", errors="ignore") def _normalize_key(name: str) -> str: s = name.lower().strip() s = s.replace("&", "and").replace("'", "").replace(".", "") s = re.sub(r"\(.*?\)", "", s) s = re.sub(r"[^a-z0-9\s-]", "", s) return re.sub(r"\s+", " ", s).strip() def _extract_hex_from_team_page(html: str) -> tuple[str, str]: """Extract primary and secondary hex codes from a teamcolorcodes.com team page.""" soup = BeautifulSoup(html, "html.parser") hexes = [] # TeamColorCodes stores swatches in .colorblock divs or inline background-color styles for block in soup.select(".colorblock, [style*='background-color']"): style = block.get("style", "") m = re.search(r"background-color:\s*(#[0-9a-fA-F]{6})", style) if m: code = m.group(1).upper() if code not in ("#FFFFFF", "#000000", "#F5F5F5", "#EEEEEE") and code not in hexes: hexes.append(code) text = block.get_text(" ", strip=True) for hm in re.findall(r"#[0-9a-fA-F]{6}", text): code = hm.upper() if code not in ("#FFFFFF", "#000000") and code not in hexes: hexes.append(code) # Fallback: scan page body text for Hex Color: #XXXXXX if len(hexes) < 2: for hm in re.findall(r"Hex(?:\s+Color)?:\s*(#[0-9a-fA-F]{6})", soup.get_text(), re.I): code = hm.upper() if code not in ("#FFFFFF", "#000000") and code not in hexes: hexes.append(code) primary = hexes[0] if len(hexes) >= 1 else "#1E293B" secondary = hexes[1] if len(hexes) >= 2 else "#64748B" return primary, secondary def scrape_ncaa_team_colors(force_refresh: bool = False) -> dict[str, dict[str, str]]: """ Scrapes NCAA team primary & secondary hex codes from https://teamcolorcodes.com/ncaa-color-codes/ and caches to team_colors.json. KAHL INDEX, R/7 & RA/7 TEAM CALCULATIONS # --------------------------------------------------------------------------- def compute_team_metrics(teams: list[dict], games: list[dict], color_map: dict) -> list[dict]: """ Computes: - R/7 (Runs Scored per 7 innings) - RA/7 (Runs Allowed per 7 innings) - SOS (Strength of Schedule via iterative opponent rating) - Kahl Index (opponent-adjusted net run differential per 7 innings + SOS) - Team Primary & Secondary Colors """ team_lookup = {t["team"]: t for t in teams} # Accumulate raw runs, runs allowed, innings, and opponent lists for t in teams: t.setdefault("runs", 0.0) t.setdefault("runs_allowed", 0.0) t.setdefault("innings_off", 0.0) t.setdefault("innings_def", 0.0) t.setdefault("wins", 0) t.setdefault("losses", 0) t["opponents"] = [] for g in games: t1, t2 = g["home"], g["away"] r1, r2 = float(g["home_runs"]), float(g["away_runs"]) inn = float(g.get("innings", 7.0)) if t1 in team_lookup and t2 in team_lookup: team_lookup[t1]["runs"] += r1 team_lookup[t1]["runs_allowed"] += r2 team_lookup[t1]["innings_off"] += inn team_lookup[t1]["innings_def"] += inn team_lookup[t1]["opponents"].append(t2) if r1 > r2: team_lookup[t1]["wins"] += 1 else: team_lookup[t1]["losses"] += 1 team_lookup[t2]["runs"] += r2 team_lookup[t2]["runs_allowed"] += r1 team_lookup[t2]["innings_off"] += inn team_lookup[t2]["innings_def"] += inn team_lookup[t2]["opponents"].append(t1) if r2 > r1: team_lookup[t2]["wins"] += 1 else: team_lookup[t2]["losses"] += 1 # Straight-up R/7 and RA/7 for t in teams: off_inn = t["innings_off"] if t["innings_off"] > 0 else max(1, (t["wins"] + t["losses"]) * 7) def_inn = t["innings_def"] if t["innings_def"] > 0 else max(1, (t["wins"] + t["losses"]) * 7) t["r_per_7"] = round((t["runs"] / off_inn) * 7.0, 2) t["ra_per_7"] = round((t["runs_allowed"] / def_inn) * 7.0, 2) t["raw_margin"] = t["r_per_7"] - t["ra_per_7"] # Iterative Kahl Index & SOS solver (25 iterations to convergence) ratings = {t["team"]: t["raw_margin"] for t in teams} sos_map = {t["team"]: 0.0 for t in teams} for _ in range(25): new_ratings = {} for t in teams: name = t["team"] opps = [o for o in t["opponents"] if o in ratings] if opps: avg_opp = sum(ratings[o] for o in opps) / len(opps) else: avg_opp = 0.0 sos_map[name] = avg_opp new_ratings[name] = t["raw_margin"] + avg_opp mean_r = sum(new_ratings.values()) / max(1, len(new_ratings)) ratings = {k: v - mean_r for k, v in new_ratings.items()} # Normalize SOS on a clean 0-100 or signed scale (signed SOS runs/7 centered at 0.00) for t in teams: name = t["team"] colors = resolve_team_colors(name, color_map) t["kahl_index"] = round(ratings[name], 2) t["sos"] = round(sos_map[name], 2) t["primary_color"] = colors["primary"] t["secondary_color"] = colors["secondary"] t["games_played"] = t["wins"] + t["losses"] t.pop("opponents", None) teams.sort(key=lambda x: x["kahl_index"], reverse=True) for idx, t in enumerate(teams, 1): t["rank"] = idx return teams # --------------------------------------------------------------------------- # 3. RC7 PLAYER METRICS (QUALIFIED ONLY, NO PA/HR IN OUTPUT) # --------------------------------------------------------------------------- def compute_player_rc7(players: list[dict], team_lookup: dict[str, dict]) -> list[dict]: """ Filters to qualified hitters (PA >= 2.0 * team games played), computes RC7 (Raw), OPAF (Opponent Pitching Adjustment Factor), and RC7 (Adj), and strips PA/HR from the final display payload. """ qualified = [] d1_avg_ra7 = ( sum(t["ra_per_7"] for t in team_lookup.values()) / max(1, len(team_lookup)) ) for p in players: team_name = p["team"] team_info = team_lookup.get(team_name) if not team_info: continue gp = team_info.get("games_played", 0) min_pa = math.ceil(gp * QUALIFIED_PA_PER_GAME) pa = int(p.get("pa", 0)) if pa < max(1, min_pa): continue # Runs Created per 21 outs (7 innings): RC = ((H + BB) * TB) / (AB + BB) ab = float(p.get("ab", 0)) h = float(p.get("h", 0)) bb = float(p.get("bb", 0)) hbp = float(p.get("hbp", 0)) tb = float(p.get("tb", 0)) outs = max(1.0, (ab - h) + float(p.get("sh", 0)) + float(p.get("sf", 0))) on_base = h + bb + hbp denom = ab + bb + hbp rc_total = (on_base * tb) / denom if denom > 0 else 0.0 rc7_raw = (rc_total / outs) * 21.0 # OPAF: Opponent Pitching Adjustment Factor based on team SOS & opponent RA/7 opp_ra7_est = max(1.5, d1_avg_ra7 - (team_info["sos"] * 0.5)) opaf = p.get("opaf") if p.get("opaf") is not None else (d1_avg_ra7 / opp_ra7_est) rc7_adj = rc7_raw * opaf qualified.append({ "player": p["player"], "team": team_name, "conference": team_info["conference"], "primary_color": team_info["primary_color"], "secondary_color": team_info["secondary_color"], "rc7_raw": round(rc7_raw, 2), "opaf": round(opaf, 3), "rc7_adj": round(rc7_adj, 2), }) qualified.sort(key=lambda x: x["rc7_adj"], reverse=True) for idx, p in enumerate(qualified, 1): p["rank"] = idx return qualified # --------------------------------------------------------------------------- # 4. 2026 NCAA SOFTBALL BRACKETOLOGY (64 TEAMS, 1-8 QUADRANT SEEDING, BUBBLE) # --------------------------------------------------------------------------- def build_2026_bracketology(teams: list[dict]) -> dict: """ Implements the 2026 NCAA Division I Softball Championship structure: - Top 32 get 1-8 regional seeds; teams 33-64 are unseeded regional pod teams top32 = field[:32] unseeded = field[32:64] def _fmt_team(t: dict, regional_seed: int | str) -> dict: return { "team": t["team"], "conference": t["conference"], "kahl_index": t["kahl_index"], "record": f"{t['wins']}-{t['losses']}", "overall_seed": t["overall_seed"], "regional_seed": regional_seed, "bid_type": t["bid_type"], "primary_color": t["primary_color"], "secondary_color": t["secondary_color"], } #
